flag_football_ep.ingest.hudl:
  GameMeta: dataclass(game_id: str, filename: str, season: int, game_date: str | None,
                      team1: str, team2: str, competition: str)
  parse_filename(path: Path) -> GameMeta            # raises FilenameError
  read_export(path: Path) -> pl.DataFrame           # separator=";", utf-8-sig, all columns as Utf8
  parse_result_tokens(df: pl.DataFrame) -> pl.DataFrame     # adds tok_* boolean columns
  ingest_file(path, contract, half_boundaries, team_mapping) -> tuple[pl.DataFrame, IngestNotices]
  ingest_dir(directory, contract, half_boundaries, team_mapping) -> list[tuple[GameMeta, pl.DataFrame, IngestNotices]]
  FilenameError(Exception)
  IngestNotices: dataclass(game_id: str, header: HeaderReport, domain: list[DomainViolation],
                           conform: ConformReport, messages: list[str])

RESULT grammar (docs/data-contract.schema.json, column RESULT):
  separator ", " · exact token match · case-sensitive
  base tokens     : Rush, KNEEL, Sack, Interception, Complete, Incomplete, Good, No Good, Fumble, Penalty
  modifier tokens : TD, Def TD, Safety, Penalty

Derived-field rules (docs/data-contract.schema.json derived_fields):
  play_id     = PLAY # (identity)
  yardline_50 = -x if x < 0 else 50 - x   (from YARD LN)
  drive_id    = increment on O/D possession flip; a scoring/turnover RESULT also closes a drive
  half        = 1 if PLAY # < half2_first_play else 2   (data/reference/half_boundaries.csv)
  posteam     = TEAM1 when ODK == "O", else TEAM2
